🔊 Web Audio API Sin archivos

Genera sonidos completamente con código JavaScript, sin necesidad de archivos MP3/WAV.

Crear contexto de audio

Requisito: Se debe llamar por primera vez dentro de un gesto del usuario (clic, tecla, toque).
let audioCtx = null;
let masterGain = null;

function getAudio() {
    if (!audioCtx) {
        const AC = window.AudioContext || window.webkitAudioContext;
        if (!AC) return null;
        audioCtx = new AC();
        masterGain = audioCtx.createGain();
        masterGain.connect(audioCtx.destination);
    }
    if (audioCtx.state === 'suspended') audioCtx.resume();
    return audioCtx;
}

Tick / clic

Sonido corto tipo "tic" para conteos, cronómetros, selecciones.

intensity: 0.0 (suave/agudo) a 1.0 (fuerte/grave).
function playTick(intensity = 0.5) {
    const ac = getAudio();
    if (!ac) return;
    const o = ac.createOscillator();
    const g = ac.createGain();
    o.type = 'square';
    o.frequency.value = 500 + 1000 * intensity;
    const t0 = ac.currentTime;
    g.gain.setValueAtTime(0.04 + 0.08 * intensity, t0);
    g.gain.exponentialRampToValueAtTime(0.0001, t0 + 0.05);
    o.connect(g);
    g.connect(masterGain);
    o.start(t0);
    o.stop(t0 + 0.06);
}

Whoosh (viento)

Ruido filtrado que simula un "whoosh" de velocidad. Ideal para animaciones de entrada/salida.

duration: duración en segundos (ej: 3.5).
function playWhoosh(duration = 3) {
    const ac = getAudio();
    if (!ac) return;
    const len = Math.max(1, Math.floor(ac.sampleRate * duration));
    const buffer = ac.createBuffer(1, len, ac.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < len; i++)
        data[i] = (Math.random() * 2 - 1) * (1 - i / len);
    const src = ac.createBufferSource();
    src.buffer = buffer;
    const filter = ac.createBiquadFilter();
    filter.type = 'bandpass';
    filter.Q.value = 0.9;
    const t0 = ac.currentTime;
    filter.frequency.setValueAtTime(250, t0);
    filter.frequency.exponentialRampToValueAtTime(3500, t0 + duration * 0.45);
    filter.frequency.exponentialRampToValueAtTime(180, t0 + duration);
    const g = ac.createGain();
    g.gain.setValueAtTime(0.0001, t0);
    g.gain.exponentialRampToValueAtTime(0.1, t0 + 0.1);
    g.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
    src.connect(filter);
    filter.connect(g);
    g.connect(masterGain);
    src.start(t0);
}

Sonido de victoria (arpegio)

Acorde ascendente para notificaciones de éxito, logros, ganadores.

function playWinSound() {
    const ac = getAudio();
    if (!ac) return;
    const notes = [523.25, 659.25, 783.99, 1046.5]; // C5 E5 G5 C6
    const t0 = ac.currentTime;
    notes.forEach((f, i) => {
        const o = ac.createOscillator();
        const g = ac.createGain();
        o.type = 'triangle';
        o.frequency.value = f;
        const start = t0 + i * 0.13;
        g.gain.setValueAtTime(0.0001, start);
        g.gain.exponentialRampToValueAtTime(0.18, start + 0.02);
        g.gain.exponentialRampToValueAtTime(0.0001, start + 0.5);
        o.connect(g);
        g.connect(masterGain);
        o.start(start);
        o.stop(start + 0.55);
    });
}

Animación con requestAnimationFrame

let rotation = 0;
let animating = false;

function animate() {
    if (!animating) return;
    rotation += 0.02;
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.save();
    ctx.translate(200, 200);
    ctx.rotate(rotation);
    ctx.fillStyle = '#e94560';
    ctx.fillRect(-50, -50, 100, 100);
    ctx.restore();
    requestAnimationFrame(animate);
}

function startAnimation() { animating = true; animate(); }
function stopAnimation() { animating = false; }

Dibujar imagen en círculo (clip)

function drawImageCircle(img, x, y, size) {
    ctx.save();
    ctx.beginPath();
    ctx.arc(x, y, size / 2, 0, Math.PI * 2);
    ctx.clip();
    ctx.drawImage(img, x - size / 2, y - size / 2, size, size);
    ctx.restore();
}

Animación easeOut (frenado suave)

// Cubic ease-out: desaceleración suave
function easeOut(progress) {
    return 1 - Math.pow(1 - progress, 3);
}

// Uso en animación:
const duration = 2000; // ms
const start = Date.now();
function animate() {
    const elapsed = Date.now() - start;
    const progress = Math.min(elapsed / duration, 1);
    const eased = easeOut(progress);
    // eased va de 0 a 1 desacelerando al final
    if (progress < 1) requestAnimationFrame(animate);
}
animate();

🎥 Grabación de video (Canvas a MP4)

Graba el contenido de un canvas y descarga el video como MP4 o WebM.

Grabación con MediaRecorder (recomendado)

let recRecorder = null;
let recChunks = [];
let recStream = null;

function startRecording(canvasElement, mime = 'video/mp4') {
    const stream = canvasElement.captureStream(30);
    recChunks = [];
    recRecorder = new MediaRecorder(stream, {
        mimeType: mime,
        videoBitsPerSecond: 12000000,
        audioBitsPerSecond: 128000  // si hay pista de audio
    });
    recRecorder.ondataavailable = (e) => {
        if (e.data && e.data.size) recChunks.push(e.data);
    };
    recRecorder.onstop = () => {
        const blob = new Blob(recChunks, { type: mime });
        recChunks = [];
        recRecorder = null;
        downloadBlob(blob, 'grabacion.' + (mime.includes('mp4') ? 'mp4' : 'webm'));
    };
    recRecorder.start(1000);
}

function stopRecording() {
    if (recRecorder && recRecorder.state !== 'inactive') recRecorder.stop();
}

function downloadBlob(blob, filename) {
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    setTimeout(() => { URL.revokeObjectURL(a.href); a.remove(); }, 1500);
}

let recCanvas = null;

function ensureRecCanvas(displayCanvas) {
    if (!recCanvas) {
        recCanvas = document.createElement('canvas');
        recCanvas.width = displayCanvas.width * REC_SCALE;
        recCanvas.height = displayCanvas.height * REC_SCALE;
    }
    const c = recCanvas.getContext('2d');
    // Copiar el canvas de pantalla al de grabación
    c.drawImage(displayCanvas, 0, 0, recCanvas.width, recCanvas.height);
    return recCanvas;
}

Detectar soporte de formato

function detectRecorder() {
    if (typeof MediaRecorder !== 'undefined') {
        if (MediaRecorder.isTypeSupported('video/mp4;codecs=avc1.42E01E') ||
            MediaRecorder.isTypeSupported('video/mp4'))
            return 'mp4-media';
    }
    if (typeof MediaRecorder !== 'undefined') {
        if (MediaRecorder.isTypeSupported('video/webm;codecs=vp9') ||
            MediaRecorder.isTypeSupported('video/webm;codecs=vp8') ||
            MediaRecorder.isTypeSupported('video/webm'))
            return 'webm-media';
    }
    return null; // no soportado
}

🎬 Grabación con audio del juego

Mezcla sonidos de WebAudio con el video del canvas en un solo archivo MP4.

// 1. Crear destino de audio para grabación
let recAudioDest = null;

function setupAudioForRecording() {
    const ac = getAudio(); // tu función getAudio
    if (!ac || !masterGain) return;
    if (!recAudioDest) {
        recAudioDest = ac.createMediaStreamDestination();
        masterGain.connect(recAudioDest); // captura todo lo que suena
    }
}

// 2. Combinar video + audio en un solo stream
function getCombinedStream(canvasElement) {
    setupAudioForRecording();
    const videoStream = canvasElement.captureStream(30);
    const audioTrack = recAudioDest
        ? recAudioDest.stream.getAudioTracks()[0]
        : null;
    return audioTrack
        ? new MediaStream([...videoStream.getVideoTracks(), audioTrack])
        : videoStream;
}

// 3. Usar el stream combinado con MediaRecorder
function startRecordingWithAudio(canvasElement) {
    const combined = getCombinedStream(canvasElement);
    recChunks = [];
    recRecorder = new MediaRecorder(combined, {
        mimeType: 'video/mp4;codecs=avc1.42E01E',
        videoBitsPerSecond: 12000000,
        audioBitsPerSecond: 128000
    });
    recRecorder.ondataavailable = (e) => {
        if (e.data && e.data.size) recChunks.push(e.data);
    };
    recRecorder.onstop = () => {
        const blob = new Blob(recChunks, { type: 'video/mp4' });
        recChunks = [];
        downloadBlob(blob, 'grabacion.mp4');
    };
    recRecorder.start(1000);
}

// 4. Limpiar al terminar
function cleanupRecording() {
    if (recRecorder && recRecorder.state !== 'inactive') recRecorder.stop();
    if (recAudioDest && masterGain) {
        try { masterGain.disconnect(recAudioDest); } catch (e) {}
        recAudioDest = null;
    }
}

🌍 Sistema de idiomas (i18n)

Traduce toda la interfaz con un objeto de traducciones y un selector de idioma.

// Objeto de traducciones
const L = {
    es: { greeting: 'Hola', goodbye: 'Adiós', action: 'Crear' },
    en: { greeting: 'Hello', goodbye: 'Goodbye', action: 'Create' },
    pt: { greeting: 'Olá', goodbye: 'Adeus', action: 'Criar' },
    fr: { greeting: 'Bonjour', goodbye: 'Au revoir', action: 'Créer' },
    de: { greeting: 'Hallo', goodbye: 'Auf Wiedersehen', action: 'Erstellen' }
};

let currentLang = 'es';

function t(key) {
    return L[currentLang]?.[key] || L['es']?.[key] || key;
}

// Detectar idioma del navegador
function detectLang() {
    const nav = (navigator.languages && navigator.languages[0])
        ? navigator.languages[0]
        : navigator.language;
    const code = (nav || 'es').split('-')[0];
    return L[code] ? code : 'es';
}

// Aplicar idioma a todos los elementos con data-i18n
function setLanguage(lang) {
    if (!L[lang]) lang = 'es';
    currentLang = lang;
    document.querySelectorAll('[data-i18n]').forEach(el => {
        el.textContent = t(el.dataset.i18n);
    });
    document.querySelectorAll('[data-i18n-placeholder]').forEach(el => {
        el.placeholder = t(el.dataset.i18nPlaceholder);
    });
    document.querySelectorAll('[data-i18n-title]').forEach(el => {
        el.title = t(el.dataset.i18nTitle);
    });
    try { localStorage.setItem('my-lang', lang); } catch (e) {}
    document.querySelectorAll('.lang-btn').forEach(btn => {
        btn.classList.toggle('active', btn.dataset.lang === lang);
    });
}

// Inicializar
const saved = localStorage.getItem('my-lang');
setLanguage(saved || detectLang());

🖼️ Procesamiento de imágenes

Redimensiona imágenes subidas para optimizar rendimiento.

const MAX_DIMENSION = 800;

function processImage(file, callback) {
    const reader = new FileReader();
    reader.onerror = () => callback(null);
    reader.onload = (e) => {
        const img = new Image();
        img.onerror = () => callback(null);
        img.onload = () => {
            const w = img.width;
            const h = img.height;
            if (w <= MAX_DIMENSION && h <= MAX_DIMENSION) {
                callback(img);
                return;
            }
            const scale = MAX_DIMENSION / Math.max(w, h);
            const cw = Math.round(w * scale);
            const ch = Math.round(h * scale);
            const c = document.createElement('canvas');
            c.width = cw;
            c.height = ch;
            c.getContext('2d').drawImage(img, 0, 0, cw, ch);
            const resized = new Image();
            resized.onload = () => { callback(resized); URL.revokeObjectURL(resized.src); };
            resized.src = c.toDataURL('image/webp', 0.85);
        };
        img.src = e.target.result;
    };
    reader.readAsDataURL(file);
}

// Crear miniatura
function makeThumb(img, size = 128) {
    const scale = size / Math.max(img.width, img.height);
    const w = Math.max(1, Math.round(img.width * scale));
    const h = Math.max(1, Math.round(img.height * scale));
    const c = document.createElement('canvas');
    c.width = w;
    c.height = h;
    c.getContext('2d').drawImage(img, 0, 0, w, h);
    return c;
}

💾 Persistencia en localStorage

Guarda y carga configuración del usuario de forma local (sin servidor).

// Guardar
function save(key, value) {
    try {
        localStorage.setItem(key, JSON.stringify(value));
        return true;
    } catch (e) {
        return false;
    }
}

// Cargar
function load(key, fallback = null) {
    try {
        const raw = localStorage.getItem(key);
        return raw !== null ? JSON.parse(raw) : fallback;
    } catch (e) {
        return fallback;
    }
}

// Eliminar
function remove(key) {
    try { localStorage.removeItem(key); } catch (e) {}
}

// Ejemplo de uso:
save('my-settings', { theme: 'noche', lang: 'es', volume: 80 });
const settings = load('my-settings', { theme: 'claro', lang: 'en' });

🎲 Aleatoriedad segura (crypto)

Genera números aleatorios criptográficamente seguros para sorteos, ruletas, etc.

// Número aleatorio entre 0 (incl.) y 1 (excl.) — seguro
function cryptoRand() {
    const buf = new Uint32Array(1);
    crypto.getRandomValues(buf);
    return buf[0] / 4294967296;
}

// Entero aleatorio entre min (incl.) y max (incl.)
function cryptoInt(min, max) {
    return min + Math.floor(cryptoRand() * (max - min + 1));
}

// Elegir elemento aleatorio de un array
function cryptoChoice(arr) {
    return arr[Math.floor(cryptoRand() * arr.length)];
}

// Mezclar array (Fisher-Yates seguro)
function cryptoShuffle(arr) {
    const a = [...arr];
    for (let i = a.length - 1; i > 0; i--) {
        const j = Math.floor(cryptoRand() * (i + 1));
        [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
}

// Ejemplo: elegir ganador de un array
const participantes = ['Ana', 'Bob', 'Carlos', 'Diana'];
const ganador = cryptoChoice(participantes);

const fbDB = firebase.firestore();

// Guardar documento
async function saveDoc(collection, docId, data) {
    try {
        await fbDB.collection(collection).doc(docId).set(data, { merge: true });
        return true;
    } catch (e) {
        console.error('Error guardando:', e);
        return false;
    }
}

// Cargar documento
async function loadDoc(collection, docId) {
    try {
        const snap = await fbDB.collection(collection).doc(docId).get();
        return snap.exists ? snap.data() : null;
    } catch (e) {
        console.error('Error cargando:', e);
        return null;
    }
}

// Cargar todos los documentos de un usuario
async function loadUserDocs(collection, uid) {
    try {
        const snap = await fbDB.collection(collection)
            .where('uid', '==', uid)
            .orderBy('createdAt', 'desc')
            .get();
        return snap.docs.map(d => ({ id: d.id, ...d.data() }));
    } catch (e) {
        console.error('Error listando:', e);
        return [];
    }
}

// Eliminar documento
async function deleteDoc(collection, docId) {
    try {
        await fbDB.collection(collection).doc(docId).delete();
        return true;
    } catch (e) {
        console.error('Error eliminando:', e);
        return false;
    }
}

// Ejemplo: guardar set de imágenes
async function saveImageSet(uid, name, images) {
    const docId = uid + '_' + Date.now();
    await saveDoc('imageSets', docId, {
        uid, name, images,
        createdAt: firebase.firestore.FieldValue.serverTimestamp()
    });
}
