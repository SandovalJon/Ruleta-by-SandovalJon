CSSJS 1. Tema oscuro / claro

Sistema de temas con variables CSS, cambio por data-theme en body y persistencia en localStorage.

CSS

:root {
  --bg: #0d1117;
  --card: #161b22;
  --accent: #58a6ff;
  --text: #c9d1d9;
}
body[data-theme="claro"] {
  --bg: #f6f8fa;
  --card: #ffffff;
  --accent: #0969da;
  --text: #1f2328;
}
body {
  background: var(--bg);
  color: var(--text);
  transition: background 0.3s, color 0.3s;
}

JS

function initTheme() {
  const saved = localStorage.getItem('theme') || 'oscuro';
  document.body.setAttribute('data-theme', saved);
}

function toggleTheme() {
  const current = document.body.getAttribute('data-theme');
  const next = current === 'oscuro' ? 'claro' : 'oscuro';
  document.body.setAttribute('data-theme', next);
  localStorage.setItem('theme', next);
}

initTheme();

let currentLang = localStorage.getItem('lang') || detectLang();

function detectLang() {
  const nav = navigator.language.slice(0, 2);
  return L[nav] ? nav : 'es';
}

function t(key) {
  return L[currentLang][key];
}

function setLanguage(lang) {
  if (!L[lang]) lang = 'es';
  currentLang = lang;
  localStorage.setItem('lang', lang);
  document.querySelectorAll('[data-i18n]').forEach(el => {
    const key = el.getAttribute('data-i18n');
    el.textContent = t(key) || key;
  });
}

setLanguage(currentLang);

JS 4. Web Audio - efectos

Efectos de sonido sintetizados con Web Audio API. Sin archivos externos, todo generado por código.

JS

let audioCtx, masterGain;

function getAudio() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    masterGain = audioCtx.createGain();
    masterGain.connect(audioCtx.destination);
    masterGain.gain.value = 0.5;
  }
  return audioCtx;
}

function playTick() {
  if (!soundEnabled) return;
  const ctx = getAudio();
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.connect(gain);
  gain.connect(masterGain);
  osc.frequency.value = 1800;
  osc.type = 'sine';
  gain.gain.setValueAtTime(0.3, ctx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.05);
  osc.start(ctx.currentTime);
  osc.stop(ctx.currentTime + 0.05);
}

function playWinSound() {
  if (!soundEnabled) return;
  const ctx = getAudio();
  [523, 659, 784, 1047].forEach((freq, i) => {
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.connect(gain);
    gain.connect(masterGain);
    osc.frequency.value = freq;
    osc.type = 'sine';
    const t = ctx.currentTime + i * 0.12;
    gain.gain.setValueAtTime(0.3, t);
    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.3);
    osc.start(t);
    osc.stop(t + 0.3);
  });
}

function playWhoosh() {
  if (!soundEnabled) return;
  const ctx = getAudio();
  const bufferSize = ctx.sampleRate * 0.3;
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = (Math.random() * 2 - 1) * (1 - i / bufferSize);
  }
  const source = ctx.createBufferSource();
  source.buffer = buffer;
  const filter = ctx.createBiquadFilter();
  filter.type = 'bandpass';
  filter.frequency.value = 800;
  source.connect(filter);
  filter.connect(masterGain);
  source.start();
}

function playErrorSound() {
  if (!soundEnabled) return;
  const ctx = getAudio();
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.connect(gain);
  gain.connect(masterGain);
  osc.frequency.value = 200;
  osc.type = 'square';
  gain.gain.setValueAtTime(0.2, ctx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.4);
  osc.start(ctx.currentTime);
  osc.stop(ctx.currentTime + 0.4);
}

function playNotif() {
  if (!soundEnabled) return;
  const ctx = getAudio();
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.connect(gain);
  gain.connect(masterGain);
  osc.frequency.value = 880;
  osc.type = 'sine';
  gain.gain.setValueAtTime(0.25, ctx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.15);
  osc.start(ctx.currentTime);
  osc.stop(ctx.currentTime + 0.15);
}

JS 6. Procesamiento de imágenes

Redimensionamiento de imágenes hasta 800px y generación de miniaturas con Canvas.

JS

function processImage(file, maxPx = 800) {
  return new Promise(resolve => {
    const reader = new FileReader();
    reader.onload = e => {
      const img = new Image();
      img.onload = () => {
        let w = img.width, h = img.height;
        if (w > maxPx || h > maxPx) {
          if (w > h) { h = h * maxPx / w; w = maxPx; }
          else { w = w * maxPx / h; h = maxPx; }
        }
        const canvas = document.createElement('canvas');
        canvas.width = w;
        canvas.height = h;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0, w, h);
        resolve(canvas.toDataURL('image/jpeg', 0.85));
      };
      img.src = e.target.result;
    };
    reader.readAsDataURL(file);
  });
}

function makeThumb(file, size = 100) {
  return processImage(file, size);
}

JS 7. localStorage

Funciones wrapper para guardar, cargar y eliminar datos en localStorage con serialización JSON.

JS

function save(key, value) {
  try {
    localStorage.setItem(key, JSON.stringify(value));
    return true;
  } catch (e) {
    console.error('Error guardando:', e);
    return false;
  }
}

function load(key, fallback = null) {
  try {
    const data = localStorage.getItem(key);
    return data !== null ? JSON.parse(data) : fallback;
  } catch (e) {
    console.error('Error cargando:', e);
    return fallback;
  }
}

function remove(key) {
  localStorage.removeItem(key);
}

JS 8. Crypto random

Generación de números aleatorios criptográficamente seguros: float, entero en rango, elección de array y shuffle Fisher-Yates.

JS

function cryptoRand() {
  const array = new Uint32Array(1);
  crypto.getRandomValues(array);
  return array[0] / 4294967296;
}

function cryptoInt(min, max) {
  return Math.floor(cryptoRand() * (max - min + 1)) + min;
}

function cryptoChoice(arr) {
  return arr[cryptoInt(0, arr.length - 1)];
}

function cryptoShuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = cryptoInt(0, i);
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

Firebase 10. Firestore

Operaciones CRUD con Firestore: guardar, cargar, listar documentos del usuario y eliminar.

JS

const db = firebase.firestore();

async function saveDoc(collection, id, data) {
  await db.collection(collection).doc(id).set(data, { merge: true });
}

async function loadDoc(collection, id) {
  const snap = await db.collection(collection).doc(id).get();
  return snap.exists ? snap.data() : null;
}

async function loadUserDocs(collection, uid) {
  const snap = await db.collection(collection)
    .where('uid', '==', uid)
    .orderBy('createdAt', 'desc')
    .get();
  return snap.docs.map(doc => ({ id: doc.id, ...doc.data() }));
}

async function deleteDoc(collection, id) {
  await db.collection(collection).doc(id).delete();
}

MediaRecorder 11. Grabar canvas a MP4

Grabación de canvas HTML a archivo usando MediaRecorder con captureStream. Inicio, parada y descarga.

JS

let mediaRecorder;
let recordedChunks = [];

function startRecording(canvas, fps = 30) {
  const stream = canvas.captureStream(fps);
  recordedChunks = [];
  mediaRecorder = new MediaRecorder(stream, {
    mimeType: 'video/webm;codecs=vp9'
  });
  mediaRecorder.ondataavailable = e => {
    if (e.data.size > 0) recordedChunks.push(e.data);
  };
  mediaRecorder.start();
}

function stopRecording() {
  return new Promise(resolve => {
    mediaRecorder.onstop = () => {
      const blob = new Blob(recordedChunks, { type: 'video/webm' });
      resolve(blob);
    };
    mediaRecorder.stop();
  });
}

async function downloadBlob(blob, filename = 'grabacion.webm') {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}

MediaRecorder 12. Grabar con audio

Grabación combinada de canvas + audio usando MediaStreamDestination conectado a masterGain.

JS

let combinedRecorder;
let combinedChunks = [];

function startRecordingWithAudio(canvas, masterGainNode, fps = 30) {
  const ctx = getAudio();
  const dest = ctx.createMediaStreamDestination();
  masterGainNode.connect(dest);

  const videoStream = canvas.captureStream(fps);
  const tracks = [
    ...videoStream.getVideoTracks(),
    ...dest.stream.getAudioTracks()
  ];
  const combinedStream = new MediaStream(tracks);

  combinedChunks = [];
  combinedRecorder = new MediaRecorder(combinedStream, {
    mimeType: 'video/webm;codecs=vp9,opus'
  });
  combinedRecorder.ondataavailable = e => {
    if (e.data.size > 0) combinedChunks.push(e.data);
  };
  combinedRecorder.start();
}

function stopRecordingWithAudio() {
  return new Promise(resolve => {
    combinedRecorder.onstop = () => {
      const blob = new Blob(combinedChunks, { type: 'video/webm' });
      resolve(blob);
    };
    combinedRecorder.stop();
  });
}

JS 13. Canvas dibujar/animar

Configuración de canvas 2D, dibujado de imagen circular con clip, función de easing y loop de animación con requestAnimationFrame.

JS

function setupCanvas(canvas, w, h) {
  canvas.width = w;
  canvas.height = h;
  return canvas.getContext('2d');
}

function drawImageCircle(ctx, img, cx, cy, radius) {
  ctx.save();
  ctx.beginPath();
  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
  ctx.clip();
  const size = radius * 2;
  ctx.drawImage(img, cx - radius, cy - radius, size, size);
  ctx.restore();
}

function easeOut(t) {
  return 1 - Math.pow(1 - t, 3);
}

function animateLoop(callback) {
  let startTime = null;
  function loop(timestamp) {
    if (!startTime) startTime = timestamp;
    const elapsed = timestamp - startTime;
    callback(elapsed, timestamp);
    requestAnimationFrame(loop);
  }
  requestAnimationFrame(loop);
}
